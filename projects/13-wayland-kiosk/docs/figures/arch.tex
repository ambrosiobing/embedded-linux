  \node[umlnote, anchor=north west, text width=76mm]

       {\textbf{Two DRM cards, and only one has connectors.} A compositor

        nothing else. \texttt{drm\_info} resolves it in seconds. The card
        NUMBERING is not stable across kernel versions, so nothing here
        hardcodes a number without checking it first.};

  \node[umlnote, anchor=north east, text width=76mm]
       at ($(hwl.south east)+(0,-5mm)$)
       {\textbf{Only the compositor opens DRM and evdev.} The client has
        neither the permission nor a reason: it receives
        \texttt{wl\_touch} and \texttt{wl\_keyboard} over a socket. An
        application reading \texttt{/dev/input} directly is a design that
        has gone wrong.};
