  %% ---------------- the instrument split ----------------
  \node[umlnote, anchor=north west, text width=128mm, align=left]
       at ($(host.south west)+(-4mm,-6mm)$)
       {\textbf{Two instruments, because one of them is blind to half the
        budget.} \texttt{systemd-analyze} reports userspace, and takes
        the firmware and loader figures from what the firmware hands it.
        The serial console with terminal timestamps is the only
        instrument for the part before userspace. That is why a project
        with a screen on it still has a serial cable in the parts list.};

  \node[umlnote, anchor=north west, text width=128mm, align=left]
       at ($(host.south west)+(-4mm,-24mm)$)
       {\textbf{The keyboard is an instrument too.} Acceptance asks that
        Tab and Enter work the buttons \emph{with the touch driver
        unloaded}, which is what proves the two input paths are
        independent rather than one path wearing two names.};
